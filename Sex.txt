local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer

task.spawn(function()
    while true do
    
        local character = player.Character
        if character and character:FindFirstChild("HumanoidRootPart") then
            local hrp = character.HumanoidRootPart
            local originalCFrame = hrp.CFrame
            
            local gameFolder = Workspace:FindFirstChild("Game")
            local coinsModel = gameFolder and gameFolder:FindFirstChild("Coins")
            
            if coinsModel then
                local foundCoin = false
                
                for _, coin in ipairs(coinsModel:GetChildren()) do
                    if coin.Name == "Coin" and coin:IsA("MeshPart") and coin.Parent then
                        foundCoin = true
                        
                        
                        hrp.CFrame = coin.CFrame
                        coin.CanTouch = true
                        coin.CanCollide = false
                        
                    
                        local startTime = tick()
                        while tick() - startTime < 3.0 do
                            
                            if not coin.Parent or not player.Character or not player.Character.PrimaryPart then
                                break
                            end
                           
                            RunService.Heartbeat:Wait()
                        end
                    end
                end
                
                if foundCoin then
                    hrp.CFrame = originalCFrame
                end
            end
        else
           
            player.CharacterAdded:Wait()
            task.wait(1)
        end
        
        task.wait(0.5)
    end
end)
